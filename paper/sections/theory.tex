\section{Adversarial Equilibrium Theory}
\label{sec:theory}
%% ===================================================================

Building on the threat model of \cref{sec:background}, we now derive the analytical machinery behind the three diagnostics promised in \cref{sec:intro}: per-edge rankings, the SVD-optimal direction, and per-node radii. The main result (\cref{thm:phase_transition}) characterizes how vulnerability evolves with perturbation magnitude, and the constrained sensitivity matrix $S_c$ provides the unified object from which the three diagnostics follow.

\begin{theorem}[Vulnerability Characterization for Contractive Implicit GNNs]
\label{thm:phase_transition}
Let $F_\theta(Z, \Ahat) = \phi(\Ahat ZW^\top + X_{\mathrm{proj}})$ be an IGNN-class operator satisfying:
\begin{enumerate}
\item[(A1)] $\phi$ is ReLU or any 1-Lipschitz activation; nonsmoothness at zero is handled via the conservative IFT~\cite{bolte2021conservative} (proof below).
\item[(A2)] $\norm{W}_2 \leq c$ via spectral normalization.
\item[(A3)] $\norm{J_z}_2 \leq \kappa < 1$ verified post-training ($\kappa{=}0.14$--$0.59$ across datasets; \cref{sec:convergence}). (A2) does not imply (A3) automatically; we report $\kappa$ alongside $\ecrit$ in our experiments so practitioners can audit. If (A3) fails, \AEGIS still returns $S_c$ rankings and SVD-optimal attacks but flags that the formal guarantees below are void.
\end{enumerate}
Define the critical budget
\begin{equation}
\ecrit = \frac{1 - \kappa}{\norm{W}_2}.
\end{equation}
Then structural perturbations $\delta \Ahat$ with $\norm{\delta \Ahat}_F = \varepsilon$ exhibit three regimes:

\noindent\textbf{(a) Subcritical} ($\varepsilon < \ecrit$): The perturbed operator remains contractive with a unique fixed point satisfying
\begin{equation}
\norm{\Delta \zstar}_F \leq \sigma_1(S) \cdot \varepsilon + O(\varepsilon^2),
\label{eq:shift_bound}
\end{equation}
where $S = (I - J_z)^{-1} J_A$ is the structural sensitivity matrix and $\sigma_1(S) \leq \norm{J_A}_{\mathrm{op}} / (1 - \kappa)$.

\noindent\textbf{(b) Critical} ($\varepsilon \to \ecrit$): For \emph{worst-case} directions that maximally increase $\norm{\Ahat{+}\delta\Ahat}_2$ (aligned with the top singular vector of $\Ahat$), $\norm{(I{-}J_z')^{-1}}_2$ diverges as $\Omega(1/(\ecrit{-}\varepsilon))$ from the Neumann lower bound $1/(1{-}\norm{J_z'}_2)$, which holds without normality assumptions on $J_z'$. Generic directions diverge more slowly because $\norm{J_z'}_2$ need not approach 1.

\noindent\textbf{(c) Supercritical} ($\varepsilon \geq \ecrit$): The sufficient contraction certificate becomes void ($\norm{J_z'}_2$ may exceed 1). The perturbed operator may still be contractive along low-sensitivity directions, but Banach no longer guarantees uniqueness and the part-(a) first-order guarantees lapse; we do not claim divergence as a generic post-threshold behavior, only that our certificate fails (empirical $\kappa_{\max}$ sweep: \cref{sec:phase_transition}).
\end{theorem}

\begin{proof}
\textbf{(a).} Under (A1), $F_\theta$ is piecewise-affine on each ReLU linear region; the activation pattern at $\zstar$ is generically stable for sufficiently small $\norm{\delta\Ahat}$ (the boundary set $\{\Ahat : \zstar(\Ahat)$ lies on a region face$\}$ has Lebesgue measure zero), so the conservative-gradient calculus of Bolte--Pauwels~\cite{bolte2021conservative} yields a conservative IFT on each region. With $J_z = \mathrm{diag}(\phi')(\Ahat \otimes W)$ and $\kappa{\leq}\norm{\Ahat}_2\norm{W}_2$, applying IFT to $G(z,\Ahat){=}z{-}F(z,\Ahat){=}0$ at $(\zstar,\Ahat)$ gives
\begin{equation}
\Delta \zstar = (I - J_z)^{-1} J_A \cdot \mathrm{vec}(\delta \Ahat) + O(\norm{\delta \Ahat}^2),
\end{equation}
with $J_A = \partial F/\partial \mathrm{vec}(A)$ and Neumann $\norm{(I-J_z)^{-1}}_2 \leq 1/(1-\kappa)$ yielding~\eqref{eq:shift_bound}. For contractivity preservation, $\norm{J_z'}_2 \leq (\norm{\Ahat}_2 + \varepsilon) \norm{W}_2$, so $\varepsilon < \ecrit$ guarantees $\norm{J_z'}_2 < 1$.

\textbf{(b)--(c).} Along the worst-case direction $\norm{J_z'}_2 \to \norm{\Ahat}_2\norm{W}_2{+}\varepsilon\norm{W}_2$, so $1{-}\norm{J_z'}_2 \geq \norm{W}_2(\ecrit{-}\varepsilon)$; the Neumann lower bound $\norm{(I{-}J_z')^{-1}}_2 \geq 1/(1{-}\norm{J_z'}_2)$ requires no normality assumption and gives the $\Omega(1/(\ecrit{-}\varepsilon))$ rate. For $\varepsilon \geq \ecrit$ Banach fails; the iteration may converge elsewhere, oscillate, or diverge.
\end{proof}

$\ecrit$ is sufficient, not necessary, and the unconstrained $\sqrt{r}$ slack is $7$--$14\times$ on 50-node subgraphs; $S_c$ closes this in practice (tightness ${\approx}1.00$ at $\varepsilon{=}0.01$; \cref{sec:cross_domain}). The Neumann series requires the operator-norm $\kappa$ rather than $\rho(J_z)$~\cite{stewart1990matrix}; the gap is the pseudospectral index $\eta$, bounded by the weight matrix alone:

\begin{observation}[Graph-Independent Nonnormality Bound]
\label{obs:eta_bound}
For $J_z = \mathrm{diag}(\phi')(\Ahat \otimes W)$ with symmetric $\Ahat$: if all activations are positive, $\eta \leq \kappa(V_W)$ where $W{=}V_W D_W V_W^{-1}$ and $\kappa(V_W){=}\norm{V_W}_2 \norm{V_W^{-1}}_2$, independent of $\Ahat$. For general ReLU patterns $\eta \leq \kappa(V_{J_z})$, which empirically stays near $\kappa(V_W)$ ($\eta{=}1.02$--$1.28$; \cref{sec:convergence}).
\end{observation}

\textit{Proof.} When $\phi'{\equiv}1$, $J_z{=}\Ahat\otimes W$ has Kronecker eigenvalues $\lambda_i(\Ahat)\lambda_j(W)$~\cite{stewart1990matrix}; symmetric $\Ahat$ gives $\kappa(U_{\Ahat}){=}1$, so the joint eigenvector matrix has condition number $\kappa(V_W)$, and the diagonalisable bound $\norm{(I{-}M)^{-1}}_2 \leq \kappa(V)/(1{-}\rho(M))$ yields $\eta \leq \kappa(V_W)$. Graph structure does not amplify nonnormality; the mild empirical $\eta$ traces to the spectral-norm regularisation of $W$.

\begin{proposition}[Maximally Sensitive First-Order Structural Perturbation]
\label{prop:attack}
The perturbation $\delta A^*$ maximizing $\norm{\Delta \zstar}$ to first order subject to $\norm{\delta A}_F \leq \varepsilon$ is $\delta A^* = \varepsilon \cdot \mathrm{reshape}(v_1, N \times N)$, where $v_1$ is the leading right singular vector of $S$. The maximum first-order equilibrium shift is $\varepsilon \cdot \sigma_1(S)$.
\end{proposition}

By the variational characterisation of $\sigma_1$, the direction is optimal for hidden-state damage and empirically aligned with classification damage (\cref{sec:adaptive}). The \emph{constrained sensitivity matrix} $S_c \in \R^{Nd \times |E|}$ has columns $[S_c]_{:,k} = S_{:,iN+j} + S_{:,jN+i}$ paired with edge basis $b_k{=}(e_i e_j^\top + e_j e_i^\top)/\sqrt{2}$ ($\norm{b_k}_F{=}1$, so $\sigma_1(S_c)$ is consistent with $\norm{\delta\Ahat}_F{=}\varepsilon$). The $N^2 \to |E|$ projection $P_c$ is the standard duplication-matrix reduction~\cite{magnus2019matrix} on the edge-supported symmetric subspace; the operational contribution is the matrix-free $S_c v$ in \cref{alg:aegis}. The envelope ratio is tight ($\approx 1.00$) at $\varepsilon{=}0.01$ and reaches $\leq 1.39$ at $\varepsilon{=}0.20$, identifying the dangerous direction rather than a global safety margin.

\begin{proposition}[Per-Node First-Order Sensitivity Radius]
\label{prop:radius}
For a linear head $f(z){=}Wz{+}b$ and node $v$ with margin $m_v{=}f_{y_v}(\zstar_v){-}\max_{c\neq y_v}f_c(\zstar_v) > 0$, the first-order sensitivity radius
\begin{equation}
r_v = \frac{m_v}{\norm{W_{y_v} - W_{c^*}}_2 \cdot \norm{S_v}_2}
\label{eq:radius}
\end{equation}
preserves the classification of $v$ for any $\norm{\delta\Ahat}_F < r_v$, where $S_v$ are the block-rows of $S$ at node $v$. Under the constrained threat model the tighter radius $r_v^{(c)} = m_v / (\norm{W_{y_v}{-}W_{c^*}}_2 \norm{S_{c,v}}_2) \geq r_v$ holds, using the block-rows of $S_c$ at node~$v$.
\end{proposition}

\textit{Proof sketch.} By Thm.~\ref{thm:phase_transition}(a), $\Delta \zstar_v \approx S_v \cdot \mathrm{vec}(\delta A)$, so $|\Delta f_{y_v}| \leq \norm{W_{y_v}{-}W_{c^*}}_2 \norm{S_v}_2 \norm{\delta A}_F$; misclassification requires $|\Delta f_{y_v}| \geq m_v$, and inverting gives $r_v$. Intuitively, $r_v$ is the margin divided by two amplification factors (decision-boundary sensitivity to hidden-state change, and equilibrium sensitivity to structural perturbation); high-margin, low-sensitivity nodes get large radii.

\begin{remark}[Sensitivity threshold semantics]
\label{rem:certificates}
$r_v$ is a first-order sensitivity threshold, not a probabilistic certificate: it indicates the perturbation scale at which the linearization predicts a classification change for node $v$, and is locally tight for small $\varepsilon$ but can be violated at larger magnitudes where higher-order terms dominate. The two approaches are complementary: randomized smoothing~\cite{bojchevski2020efficient} certifies which nodes are guaranteed safe, while \AEGIS differentiates which edges and nodes are most sensitive and in which direction (\cref{sec:certificates}). Empirically, breach rate is $0\%$ at $\varepsilon{=}0.01$ and rises with $\varepsilon$ as expected for first-order approximations (\cref{sec:adaptive}).
\end{remark}

\subsection{Continuous-to-Discrete Transfer}

\begin{proposition}[Continuous-to-Discrete Ranking Transfer]
\label{prop:transfer}
Under (A1)--(A3), let $w_k = [\Ahat]_{ij}$ denote the normalized weight of edge $k = (i,j) \in E$, $v_k = \norm{[S_c]_{:,k}}_2$ the continuous vulnerability score, and $d_k = \norm{\zstar(A) - \zstar(A \setminus k)}_2$ the discrete damage from full removal of edge~$k$. Assume all single-edge removals are subcritical: $\sqrt{2} \max_k w_k < \ecrit$.

\noindent\textbf{(a) First-order bridge.} The discrete damage satisfies
\begin{equation}
d_k = w_k \cdot v_k + R_k, \quad |R_k| \leq \frac{L_J}{2(1-\kappa)^2} \cdot w_k^2,
\label{eq:transfer}
\end{equation}
where $L_J$ is the Lipschitz constant of the state Jacobian $J_z$ with respect to $\mathrm{vec}(A)$.

\noindent\textbf{(b) Ranking preservation.} For two edges $k_1, k_2$ with $v_{k_1} > v_{k_2}$ and $w_{k_1} \geq w_{k_2}$, we have $d_{k_1} > d_{k_2}$ whenever
\begin{equation}
w_{k_1} < \frac{v_{k_1} - v_{k_2}}{L_J / (1-\kappa)^2}.
\label{eq:ranking}
\end{equation}
Under uniform weights ($w_k = w$ for all $k$), $v_{k_1} > v_{k_2}$ implies $d_{k_1} > d_{k_2}$ whenever $w < \min_{k_1 \neq k_2} (v_{k_1} - v_{k_2}) \cdot (1-\kappa)^2 / L_J$.
\end{proposition}

\begin{proof}
\textbf{(a).} Edge removal corresponds to $[\delta\Ahat]_{ij}{=}[\delta\Ahat]_{ji}{=}{-}w_k$, so by the $S_c$ construction $\norm{\Delta \zstar} \approx w_k v_k$. For the remainder, let $g(t){=}\zstar(A{+}t\delta\Ahat_k)$, $t{\in}[0,1]$. IGNN is piecewise affine in $A$ for a fixed activation pattern, so $g(t)$ is piecewise linear and crosses finitely many activation boundaries; on each segment $J_z$ is constant. Across a boundary $J_z$ jumps by at most $\norm{W}_2^2$ (the max change in $\mathrm{diag}(\phi')(\Ahat\otimes W)$ when one activation flips), so the worst-case path Lipschitz constant $L_J \leq \norm{W}_2^2$. The resolvent identity $\frac{d}{dt}(I-M)^{-1}{=}(I-M)^{-1}\frac{dM}{dt}(I-M)^{-1}$ then yields $\norm{g''(\xi)} \leq (1{-}\kappa)^{-2}L_J w_k^2$, hence $|R_k| \leq L_J w_k^2/(2(1{-}\kappa)^2)$.
\textbf{(b).} From (a), $d_{k_1}{-}d_{k_2} \geq w_{k_1}v_{k_1}{-}w_{k_2}v_{k_2}{-}C(w_{k_1}^2{+}w_{k_2}^2)$ with $C{=}L_J/(2(1-\kappa)^2)$; under $v_{k_1}{>}v_{k_2}$, $w_{k_1}{\geq}w_{k_2}$, the first term is positive and~\eqref{eq:ranking} ensures the first-order gap dominates.
\end{proof}

Empirically $|R_k|$ is $2$--$10\times$ smaller than $L_J w_k^2$, and the $29/33$ positive-transfer cells of \cref{fig:tau_heatmap} confirm the piecewise structure does not erode the bridge: shallow models with near-uniform $v_k$ (GCN-2) violate part-(b), deeper models satisfy it. The ingredients of \cref{thm:phase_transition} are classical~\cite{stewart1990matrix,trefethen2005spectra}; the contribution is the constrained projection $S_c$ that turns a vacuous bound (tightness $0.31$) into a tight one ($1.00$; \cref{sec:cross_domain}) and unifies attacks, rankings, and radii.

\subsection{Generalization to Explicit GNNs}

\begin{observation}[Structural Sensitivity for $K$-Layer GNNs]
\label{prop:explicit}
Let $Z_K = g_K \circ \cdots \circ g_1(X, A)$ be a $K$-layer GNN where each layer $g_l$ is differentiable with respect to both its input and $A$. Define the \emph{unrolled sensitivity matrix}
\begin{equation}
S_K = \frac{\partial \mathrm{vec}(Z_K)}{\partial \mathrm{vec}(A)} = \sum_{l=1}^{K} \left(\prod_{k=l+1}^{K} J_z^{(k)}\right) J_A^{(l)},
\label{eq:unrolled}
\end{equation}
where $J_z^{(k)} = \partial g_k / \partial Z_{k-1}$ and $J_A^{(l)} = \partial g_l / \partial \mathrm{vec}(A)$. Then:

\noindent\textbf{(a)} The first-order shift bound $\norm{\Delta Z_K}_F \leq \sigma_1(S_K) \cdot \norm{\delta A}_F + O(\norm{\delta A}^2)$ holds, with
\begin{equation}
\sigma_1(S_K) \leq \sum_{l=1}^{K} \left(\prod_{k=l+1}^{K} \norm{J_z^{(k)}}_2\right) \norm{J_A^{(l)}}_2.
\end{equation}

\noindent\textbf{(b)} The constrained construction $S_c$ and per-node radius $r_v$ (\cref{prop:radius}) apply identically with $S_K$ in place of $S$.
\end{observation}

For weight-tied models, $\sigma_1(S_K) \leq \norm{J_A}_2 (1{-}\kappa^K)/(1{-}\kappa)$ converges to the IGNN bound as $K{\to}\infty$. Explicit GNNs inherit the practical tools but lack $\ecrit$ and the regime characterisation, which depend on (A3); a Jacobian product $\prod_k \norm{J_z^{(k)}}_2 > 10$ diagnoses unreliable first-order predictions, and the empirical tightness $0.99$--$1.02$ (\cref{sec:explicit_extension}) validates the approximation regardless of the a priori bound.
