\section{Adversarial Equilibrium Theory}
\label{sec:theory}
%% ===================================================================

\Cref{sec:framework} read three diagnostics off $S_c$; we now justify them, starting with the safety boundary. The picture is a phase transition in $\varepsilon$: while $\varepsilon$ is small the model stays contractive and its prediction moves boundedly with the graph; as $\varepsilon$ nears the critical budget $\ecrit$ the equilibrium grows arbitrarily sensitive; past $\ecrit$ contraction fails and the prediction can break. \Cref{thm:phase_transition} makes this precise and puts $\ecrit$ in closed form.

\begin{theorem}[Vulnerability Characterization for Contractive Implicit GNNs]
\label{thm:phase_transition}
Let $F_\theta(Z, \Ahat) = \phi(\Ahat ZW^\top + X_{\mathrm{proj}})$ be an IGNN-class operator satisfying:
\begin{enumerate}
\item[(A1)] $\phi$ is ReLU or any 1-Lipschitz activation; nonsmoothness at zero is handled via the conservative IFT~\cite{bolte2021conservative} (proof below).
\item[(A2)] $\norm{W}_2 \leq c$ via spectral normalization.
\item[(A3)] $\norm{J_z}_2 \leq \kappa < 1$ verified post-training ($\kappa{=}0.14$--$0.59$ across our suite; \cref{tab:cross_domain}). (A2) does not imply (A3); we report the trained $\kappa$ so practitioners can audit, and \AEGIS still returns rankings and attacks if (A3) fails, its guarantees then scoped to the contractive regime.
\end{enumerate}
Define the critical budget
\begin{equation}
\ecrit = \frac{1 - \kappa}{\norm{W}_2}.
\end{equation}
Then structural perturbations $\delta \Ahat$ with $\norm{\delta \Ahat}_F = \varepsilon$ exhibit three regimes:

\noindent\textbf{(a) Subcritical} ($\varepsilon < \ecrit$): The perturbed operator remains contractive with a unique fixed point satisfying
\begin{equation}
\norm{\Delta \zstar}_F \leq \sigma_1(S) \cdot \varepsilon + O(\varepsilon^2),
\label{eq:shift_bound}
\end{equation}
where $S = (I - J_z)^{-1} J_A$ is the structural sensitivity matrix and $\sigma_1(S) \leq \norm{J_A}_{\mathrm{op}} / (1 - \kappa)$.

\noindent\textbf{(b) Critical} ($\varepsilon \to \ecrit$): The resolvent obeys $\norm{(I{-}J_z')^{-1}}_2 \geq 1/\min_i|1{-}\lambda_i(J_z')|$ and blows up when an eigenvalue of $J_z'$ approaches $1$ (a vanishing spectral gap to $1$, not $\norm{J_z'}_2\!\to\!1$). The rate is $\Omega(1/(\ecrit{-}\varepsilon))$ for normal $J_z'$ with a dominant Perron eigenvalue; in general $\ecrit$ \emph{lower-bounds} the divergence threshold, the slack being the nonnormality $\eta$ ($\eta{\leq}2.47$ for ReLU; \cref{obs:eta_bound,rem:eta_relu}).

\noindent\textbf{(c) Supercritical} ($\varepsilon \geq \ecrit$): The contraction certificate no longer applies ($\norm{J_z'}_2$ may exceed 1); $\varepsilon{<}\ecrit$ is a sufficient safety boundary that trained IGNNs sit well inside ($2$--$4\times$ spectral-radius margin to $\rho{=}1$, \cref{app:phase_scal}).
\end{theorem}

\noindent\textit{Proof in \cref{app:proof_phase}.} The subcritical bound is a conservative IFT~\cite{bolte2021conservative} with Neumann $\norm{(I{-}J_z)^{-1}}_2\le1/(1{-}\kappa)$; the critical rate follows from the resolvent identity.

The closed-form $\ecrit$ is only \emph{sufficient}; the data-dependent $S_c$ tightens it by $7$--$14\times$. The slack is the Neumann-versus-spectral-radius gap, the pseudospectral index $\eta$, which for symmetric $\Ahat$ is controlled by $W$ alone ($\eta\in[1.19,2.47]$ for ReLU; \cref{obs:eta_bound,rem:eta_relu}).

\Cref{thm:phase_transition}(a) certifies safety \emph{below} $\ecrit$; the same operator also bounds the boundary from \emph{above}, through the nonnormality $g_W{=}\norm{W}_2/\rho(W){\geq}1$ of $W$.

\begin{theorem}[Constant-factor two-sided boundary, all-active]
\label{thm:cf2s}
Under \textnormal{(A1)--(A3)} with $\Ahat$ symmetric and $\ecrit{>}0$, the all-active contraction boundary $\varepsilon_{\mathrm{br}}$ (the smallest feasible $\norm{\delta\Ahat}_F$ at which $\rho(\Ahat'{\otimes}W){\geq}1$) obeys
\begin{equation}
\ecrit \;\leq\; \varepsilon_{\mathrm{br}} \;\leq\; \tfrac{C}{\beta}\,\ecrit,
\qquad C = g_W\,\tfrac{1+\kappa}{1-\kappa},
\label{eq:bracket}
\end{equation}
with $C$ dimension-free and computable post-training, and $\beta{\in}(0,1]$ the edge-supported Frobenius mass of $\Ahat$'s Perron mode (positive for any connected graph). The upper side is attained by a rank-one critical-driving attack aligned with that mode; the bracket collapses to equality iff $g_W{=}1$ and $\beta{=}1$.
\end{theorem}

The bracket constant is loose (${\sim}10$--$16\times$), but it exposes the coupling: one operator $S_c$ supplies the certified budget $\ecrit$, the matching attack that approaches it, and, through $\sigma_1(S_c)$, the defense. Empirically the norm certificate under-states the true break by $\mathbf{2}$--$\mathbf{9\times}$ ($10$ seeds); proof and nonlinear extension in \cref{app:bracket}.

The per-node radius $r_v$ (\cref{sec:framework}) carries a first-order safety guarantee.

\begin{proposition}[Per-Node First-Order Sensitivity Radius]
\label{prop:radius}
For a linear head $f(z){=}Wz{+}b$ and a node $v$ classified as $y_v$ with positive margins $m_v^{(c)}{=}f_{y_v}(\zstar_v){-}f_c(\zstar_v){>}0$ to every competitor $c{\neq}y_v$, the first-order sensitivity radius
\begin{equation}
r_v = \min_{c\neq y_v} \frac{m_v^{(c)}}{\norm{(W_{y_v}{-}W_c)\,S_v}_2}
\label{eq:radius}
\end{equation}
preserves the classification of $v$ for any $\norm{\delta\Ahat}_F < r_v$, where $S_v$ are the block-rows of $S$ at node $v$; substituting $S_{c,v}$ gives the tighter constrained radius $r_v^{(c)}{\geq}r_v$. Minimizing over \emph{all} competitors is the distance to the nearest first-order boundary, so the bound holds even when the runner-up changes.
\end{proposition}

\noindent\textit{Proof in \cref{app:proof_radius}.} A Cauchy--Schwarz bound on each per-competitor margin change gives $r_v$ as the smallest margin divided by two amplification factors (decision-boundary and equilibrium sensitivity); high-margin, low-sensitivity nodes get large radii.

\subsection{\AEGIS-Conformal: A Distribution-Free Certificate}
\label{sec:conformal}
The radius $r_v$ of \cref{eq:radius} is a first-order threshold: locally tight, but liable to be violated at larger $\varepsilon$. We now upgrade it to a distribution-free guarantee, one that is sound under an exchangeability condition and gated by an explicit coverage test (\cref{app:conformal}). The starting point is split conformal prediction, which returns a set $C(v)$ with $\Pr[y_v\in C(v)]\geq1{-}\alpha$ on clean inputs from a \emph{conformity score}; we report the two standard choices, \emph{TPS} (the softmax probability of the candidate label;~\citealp{sadinle2019least}) and \emph{APS} (its adaptive cumulative variant;~\citealp{romano2020classification}), where APS yields slightly larger sets and TPS can abstain with an empty one. Making this hold \emph{robustly} over the ball $\norm{\delta\Ahat}_F\leq\varepsilon$ requires bounding the worst-case shift in the conformity scores, and that is exactly what the same $S_c$ sensitivity delivers in closed form: a curvature-corrected bound $L_1^{c}\varepsilon{+}C_v\varepsilon^2$ that replaces the ${\sim}10^4$-sample smoothing the task would otherwise demand. Substituting it into the binary-certificate split conformal of \citet{zargarbashi2023conformal} yields a robust set with
\begin{equation}
\Pr\!\big[y_v\in C_\varepsilon(v)\big]\;\geq\;1-\alpha
\qquad\text{for all }\ \norm{\delta\Ahat}_F\leq\varepsilon .
\label{eq:conformal}
\end{equation}
\Cref{sec:certify} validates this certificate empirically, checking both nominal-level coverage and the coverage gate under the worst-case attack, and \cref{sec:defense} returns to the coupled $\sigma_1(S_c)$ defense.

\subsection{Transfer to Discrete Edits and Explicit GNNs}
\label{sec:transfer_explicit}
Two bridges carry $S_c$ from the continuous, implicit setting to the discrete, architecture-general one practitioners face; both are proved in \cref{app:transfer}. The first reads a discrete edge-removal ranking off the continuous score. Deleting an edge is, to first order, a continuous perturbation whose size is the edge's own weight, so its damage should track the score $S_c$ already assigns it.

\begin{proposition}[Continuous-to-discrete transfer]
\label{prop:transfer}
Under \textnormal{(A1)--(A3)}, let edge $k$ have weight $w_k$ and score $v_k{=}\norm{[S_c]_{:,k}}_2$, and let $d_k$ be the true damage of removing it. If every single-edge removal is subcritical, then
\begin{equation}
d_k = w_k v_k + R_k, \qquad |R_k|\le \tfrac{L_J}{2(1-\kappa)^2}\,w_k^2,
\label{eq:transfer}
\end{equation}
with curvature constant $L_J\le\norm{W}_2^2\norm{\zstar}$. The edge-weighted score $w_k v_k$ therefore preserves the brute-force removal order whenever the score gap between two edges beats this quadratic remainder.
\end{proposition}

The signal is $O(w_k)$ and the error only $O(w_k^2)$, so the ranking is reliable for the small-weight edges of sparse graphs; empirically $w_k v_k$ reaches $\tau{=}\mathbf{+0.996}$ against brute-force N-1 on full-graph Amazon Photo (\cref{sec:explicit_extension}). The second bridge drops the implicit-model assumption: an explicit $K$-layer GNN has no fixed point, but unrolling its layers yields the same kind of structural Jacobian.

\begin{proposition}[Explicit $K$-layer GNNs]
\label{prop:explicit}
For $Z_K{=}g_K\!\circ\!\cdots\!\circ\! g_1(X,A)$ with per-layer Jacobians $J_z^{(k)}{=}\partial g_k/\partial Z_{k-1}$ and $J_A^{(l)}{=}\partial g_l/\partial\mathrm{vec}(A)$, the unrolled sensitivity
\begin{equation}
S_K=\frac{\partial\,\mathrm{vec}(Z_K)}{\partial\,\mathrm{vec}(A)}=\sum_{l=1}^{K}\Big(\prod_{k=l+1}^{K}J_z^{(k)}\Big)J_A^{(l)}
\label{eq:unrolled}
\end{equation}
replaces $S$ throughout: the constrained operator $S_c$, the per-node radius $r_v$, and the first-order shift bound $\norm{\Delta Z_K}_F\le\sigma_1(S_K)\,\norm{\delta A}_F$ all carry over unchanged.
\end{proposition}

So six of our seven architectures inherit the full audit/certify/defend pipeline from $S_K$; only the closed-form boundary $\ecrit$ stays restricted to the contractive implicit subclass.
