% --- Dedicated appendix Table of Contents (black, AAAI-clean) ------------------
% aaai2026.sty nullifies \addcontentsline (its line 44) to forbid a document ToC.
% We restore the standard kernel definition HERE, after the body has been typeset,
% so that ONLY the appendix sections below log to the .toc; the body, already
% typeset under the no-op definition, contributes nothing, giving an appendix-only
% ToC with no hyperref/xcolor. \@starttoc inputs/reopens the .toc as \tableofcontents
% does but without its built-in heading, so we control the "Appendix" /
% "Table of Contents" titles and the framing rules to match the reference layout.
% The article defaults render lettered \section entries in bold (no leaders) and
% \subsection entries with dotted leaders and page numbers. Two passes settle pages.
% \onecolumn makes the ToC span the full page width (both columns); \twocolumn below
% resumes the normal two-column layout for the appendix body.
\onecolumn
\makeatletter
% Modern LaTeX \contentsline takes FOUR arguments (the 4th is the hyperref anchor,
% empty when hyperref is absent); we write that empty {} so each entry is self-
% contained and does not swallow the next.
\renewcommand{\addcontentsline}[3]{%
  \addtocontents{#1}{\protect\contentsline{#2}{#3}{\thepage}{}}}
\makeatother
